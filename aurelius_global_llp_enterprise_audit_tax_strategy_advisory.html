Step 2: Run One-Time SQL Setup in Supabase

Paste this into your Supabase SQL Editor to create the tables and public audit-workpapers storage bucket.

-- 1. Create Engagements, Workpapers & RFPs tables
create table if not exists aurelius_engagements (
  id text primary key,
  client_name text not null,
  category text,
  period text,
  industry text,
  fee_usd numeric,
  payload jsonb,
  updated_at timestamptz default now()
);

create table if not exists aurelius_workpapers (
  id text primary key,
  client_id text not null,
  step_num int not null,
  doc_key text not null,
  file_name text not null,
  file_size text,
  storage_url text,
  uploaded_at timestamptz default now()
);

create table if not exists aurelius_rfps (
  ref_id text primary key,
  org_name text not null,
  service_line text,
  fee_estimate text,
  exec_name text,
  exec_email text,
  created_at timestamptz default now()
);

-- 2. Enable open RLS policies for demo / client intake
alter table aurelius_engagements enable row level security;
alter table aurelius_workpapers enable row level security;
alter table aurelius_rfps enable row level security;

create policy "Public access engagements" on aurelius_engagements for all using (true) with check (true);
create policy "Public access workpapers" on aurelius_workpapers for all using (true) with check (true);
create policy "Public access rfps" on aurelius_rfps for all using (true) with check (true);

-- 3. Create Storage Bucket for binary PDF/CSV working papers
insert into storage.buckets (id, name, public) values ('audit-workpapers', 'audit-workpapers', true) on conflict do nothing;
create policy "Public storage access" on storage.objects for all using (bucket_id = 'audit-workpapers') with check (bucket_id = 'audit-workpapers');
